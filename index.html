        async function fetchGeminiWithBackoff(payload, maxRetries = 3) {
            const apiKey = "";
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

            let delay = 1000;
            for (let attempt = 0; attempt <= maxRetries; attempt++) {
                try {
                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });
                    if (!response.ok) {
                        throw new Error(`HTTP ${response.status}`);
                    }
                    return await response.json();
                } catch (err) {
                    if (attempt === maxRetries) throw err;
                    await new Promise(res => setTimeout(res, delay));
                    delay *= 2;
                }
            }
        }

        // Chuyển đổi Markdown cơ bản sang HTML đẹp mắt (an toàn XSS)
        function renderFormattedMarkdown(container, rawText) {
            const escaped = rawText
                .replace(/&/g, '&')
                .replace(//g, '>');

            const html = escaped
                .replace(/^### (.*$)/gim, '
$1
')
                .replace(/^## (.*$)/gim, '
$1
')
                .replace(/\*\*(.*?)\*\*/g, '$1')
                .replace(/\*(.*?)\*/g, '$1')
                .replace(/^- (.*$)/gim, '
•$1
')
                .replace(/\n\n/g, '
')
                .replace(/\n/g, '
');

            container.innerHTML = `
${html}
`;
        }

        function initWinLossTracker() {

                    const parsed = JSON.parse(saved);
                    if (Array.isArray(parsed) && parsed.length > 0) {

        // Hàm tự động đối soát bộ số dự đoán với kết quả thực tế (GĐB & danh sách lô về)
        function autoEvaluateRecord(rec, actualGdb, actualLotoArr) {
            const cleanGdb = (actualGdb || '').replace(/\D/g, '');
            const de2 = cleanGdb.length >= 2 ? cleanGdb.slice(-2) : '';
            const lotoList = [...actualLotoArr];
            if (de2 && !lotoList.includes(de2)) {
                lotoList.push(de2);
            }

            if (!de2 && lotoList.length === 0) {
                return {
                    ...rec,
                    actualGdb: cleanGdb,
                    actualLoto: lotoList,
                    status: 'PENDING',
                    hitDetails: 'Đang chờ kết quả 18:15',
                    hitBtl: false,
                    hitStl: false,
                    hitXien: false,
                    hitDe: false
                };
            }

            const countHits = (num) => lotoList.filter(x => x === num).length;

            const btlHits = rec.btl ? countHits(rec.btl) : 0;
            const hitBtl = btlHits > 0;

            const stlHitsList = (rec.stl || []).filter(n => countHits(n) > 0);
            const hitStl = stlHitsList.length > 0;

            const xienHitsList = (rec.xien || []).filter(n => countHits(n) > 0);
            const hitXien = xienHitsList.length >= 2;

            const hitDe = de2 && Array.isArray(rec.danDe) && rec.danDe.includes(de2);

            const details = [];
            if (hitBtl) details.push(`Nổ BTL ${rec.btl}${btlHits > 1 ? ` (${btlHits} nháy)` : ''}`);
            else if (hitStl) details.push(`Nổ STL ${stlHitsList.join(', ')}`);

            if (hitXien) details.push(`Trúng Xiên ${xienHitsList.length} (${xienHitsList.join('-')})`);
            if (hitDe) details.push(`Nổ Đề ${de2}`);

            const isWin = hitBtl || hitStl || hitXien || hitDe;

            return {
                ...rec,
                actualGdb: cleanGdb,
                actualLoto: lotoList,
                status: isWin ? 'WIN' : 'LOSS',
                hitDetails: isWin ? details.join(' • ') : 'Không trúng bộ số trọng tâm trong kỳ',
                hitBtl,
                hitStl,
                hitXien,
                hitDe
            };
        }

        // Cập nhật toàn bộ Thẻ KPI & Bảng Lịch Sử
        function renderWinLossTracker() {
            const settled = trackerRecords.filter(r => r.status === 'WIN' || r.status === 'LOSS');
            const wins = trackerRecords.filter(r => r.status === 'WIN');
            const losses = trackerRecords.filter(r => r.status === 'LOSS');
            const pendings = trackerRecords.filter(r => r.status === 'PENDING');

            // Đếm trên nút filter
            document.getElementById('count-all').textContent = trackerRecords.length;
            document.getElementById('count-win').textContent = wins.length;
            document.getElementById('count-loss').textContent = losses.length;
            document.getElementById('count-pending').textContent = pendings.length;

            // Tính Win Rate %
            const winRate = settled.length > 0 ? (wins.length / settled.length) * 100 : 0;
            const winRateStr = `${winRate.toFixed(1)}%`;
            document.getElementById('kpi-win-rate').textContent = winRateStr;
            document.getElementById('kpi-win-ratio-sub').textContent = `${wins.length} kỳ trúng / ${settled.length} kỳ đã quay`;
            document.getElementById('kpi-gauge-text').textContent = `${Math.round(winRate)}%`;

            // Circular SVG Gauge (chu vi r=26 => 2 * pi * 26 = 163.36)
            const circumference = 163.36;
            const offset = circumference - (winRate / 100) * circumference;
            const gaugeCircle = document.getElementById('kpi-gauge-circle');
            gaugeCircle.style.strokeDashoffset = offset;
            gaugeCircle.style.stroke = winRate >= 55 ? '#10b981' : winRate >= 40 ? '#facc15' : '#ef4444';

            // Số kỳ Trúng vs Không Trúng
            document.getElementById('kpi-total-settled').textContent = `${settled.length} Kỳ`;
            document.getElementById('kpi-win-count').textContent = wins.length;
            document.getElementById('kpi-loss-count').textContent = losses.length;

            // Chuỗi thắng dài nhất (Max Win Streak)
            let maxStreak = 0;
            let currentStreak = 0;
            // Duyệt từ cũ đến mới
            for (let i = settled.length - 1; i >= 0; i--) {
                if (settled[i].status === 'WIN') {
                    currentStreak++;
                    if (currentStreak > maxStreak) maxStreak = currentStreak;
                } else {
                    currentStreak = 0;
                }
            }
            document.getElementById('kpi-max-streak').innerHTML = ` ${maxStreak} kỳ`;

            // Phong độ 7 kỳ gần nhất đã có kết quả
            const recent7 = settled.slice(0, 7).reverse();
            const dotsEl = document.getElementById('kpi-recent-dots');
            if (recent7.length === 0) {
                dotsEl.innerHTML = `Chưa có kỳ đối soát`;
            } else {
                dotsEl.innerHTML = recent7.map(r => {
                    const isW = r.status === 'WIN';
                    return `
                        
                            ${isW ? 'W' : 'L'}
                        
                    `;
                }).join('');
            }

            // Tỷ lệ trúng theo từng hạng mục
            const calcCatRate = (key) => {
                if (settled.length === 0) return '0%';
                const hitCount = settled.filter(r => r[key]).length;
                return `${Math.round((hitCount / settled.length) * 100)}%`;
            };
            document.getElementById('kpi-cat-btl').textContent = calcCatRate('hitBtl');
            document.getElementById('kpi-cat-stl').textContent = calcCatRate('hitStl');
            document.getElementById('kpi-cat-xien').textContent = calcCatRate('hitXien');
            document.getElementById('kpi-cat-de').textContent = calcCatRate('hitDe');

            // Lọc danh sách theo trạng thái & từ khóa tìm kiếm
            const keyword = (document.getElementById('tracker-search-input').value || '').trim().toLowerCase();
            const filtered = trackerRecords.filter(r => {
                if (trackerFilter !== 'ALL' && r.status !== trackerFilter) return false;
                if (keyword) {
                    const haystack = `${r.date} ${r.btl} ${(r.stl || []).join(' ')} ${(r.xien || []).join(' ')} ${r.actualGdb} ${r.hitDetails}`.toLowerCase();
                    if (!haystack.includes(keyword)) return false;
                }
                return true;
            });

            const tbody = document.getElementById('win-loss-tbody');
            if (filtered.length === 0) {
                tbody.innerHTML = `
                    
                        
                            Không tìm thấy bản ghi đối soát nào phù hợp với bộ lọc hiện tại.
                        
                    
                `;
                lucide.createIcons();
                return;
            }

            tbody.innerHTML = filtered.map(rec => {
                const actualSet = new Set(rec.actualLoto || []);
                if (rec.actualGdb && rec.actualGdb.length >= 2) {
                    actualSet.add(rec.actualGdb.slice(-2));
                }

                // Highlight số nào trúng trong dàn dự đoán
                const renderNumBadge = (num, isPrimary = false) => {
                    const isHit = rec.status === 'WIN' && actualSet.has(num);
                    if (isHit) {
                        return `${num}`;
                    }
                    return `${num}`;
                };

                // Huy hiệu trạng thái ĐÃ TRÚNG / KHÔNG TRÚNG / CHỜ QUAY
                let statusBadge = '';
                let rowBg = 'hover:bg-zinc-900/50';
                if (rec.status === 'WIN') {
                    rowBg = 'bg-emerald-950/15 hover:bg-emerald-950/25';
                    statusBadge = `
                        
                             ĐÃ TRÚNG
                        
                    `;
                } else if (rec.status === 'LOSS') {
                    rowBg = 'bg-red-950/10 hover:bg-red-950/20';
                    statusBadge = `
                        
                             KHÔNG TRÚNG
                        
                    `;
                } else {
                    statusBadge = `
                        
                             CHỜ 18:15
                        
                    `;
                }

                const deActual = rec.actualGdb && rec.actualGdb.length >= 2 ? rec.actualGdb.slice(-2) : '';
                const gdbDisplay = rec.actualGdb
                    ? `${rec.actualGdb.slice(0, -2)}${deActual}`
                    : `Chưa có KQ`;

                return `
                    
                        
                            
${rec.date}

                            
AI Chấm: ${rec.prob}%

                        

                        
                            

                                BTL:
                                ${renderNumBadge(rec.btl, true)}
                                STL:
                                ${(rec.stl || []).map(n => renderNumBadge(n)).join(' ')}
                            

                            

                                Dàn Đề:
                                ${(rec.danDe || []).slice(0, 6).map(n => renderNumBadge(n)).join(' ')}
                            

                        

                        
                            
ĐB: ${gdbDisplay}

                            ${rec.actualLoto && rec.actualLoto.length > 0
                                ? `
Lô: ${rec.actualLoto.slice(0, 6).join(', ')}...
`
                                : ''}
                        

                        
                            

                                ${rec.hitDetails}
                            

                        

                        
                            ${statusBadge}
                        

                        
                            

                                
                                
                                
                            

                        
                    
                `;
            }).join('');

            lucide.createIcons();
        }

        function setTrackerFilter(filter) {

            document.querySelectorAll('.tfilter-btn').forEach(btn => {

        function quickToggleRecordStatus(id) {
            const idx = trackerRecords.findIndex(r => r.id === id);
            if (idx === -1) return;
            const rec = trackerRecords[idx];
            if (rec.status === 'PENDING' || rec.status === 'LOSS') {
                rec.status = 'WIN';
                rec.hitBtl = true;
                rec.hitStl = true;
                rec.hitDetails = `Đã xác nhận TRÚNG BTL ${rec.btl}`;
            } else {
                rec.status = 'LOSS';
                rec.hitBtl = false;
                rec.hitStl = false;
                rec.hitXien = false;
                rec.hitDe = false;
                rec.hitDetails = 'Đã xác nhận KHÔNG TRÚNG';
            }
            saveTrackerToStorage();
            renderWinLossTracker();
            showToast(`Đã cập nhật trạng thái ngày ${rec.date} thành ${rec.status === 'WIN' ? 'ĐÃ TRÚNG' : 'KHÔNG TRÚNG'}!`);
        }

        function handleTrackerFormSubmit(e) {
            e.preventDefault();
            const editId = document.getElementById('edit-record-id').value;
            const date = document.getElementById('form-date').value.trim();
            const prob = parseFloat(document.getElementById('form-prob').value) || 30.0;
            const btl = pad2(parseInt(document.getElementById('form-btl').value.replace(/\D/g, ''), 10) % 100);
            const stl = (document.getElementById('form-stl').value.match(/\d{2}/g) || [btl]).map(n => pad2(parseInt(n, 10) % 100));
            const xien = (document.getElementById('form-xien').value.match(/\d{2}/g) || []).map(n => pad2(parseInt(n, 10) % 100));
            const danDe = (document.getElementById('form-dande').value.match(/\d{2}/g) || []).map(n => pad2(parseInt(n, 10) % 100));
            const actualGdb = document.getElementById('form-actual-gdb').value.replace(/\D/g, '');
            const actualLoto = (document.getElementById('form-actual-loto').value.match(/\d{2}/g) || []).map(n => pad2(parseInt(n, 10) % 100));

            const baseRec = {
                id: editId || 'rec-' + Date.now(),
                date,
                prob,
                btl,
                stl,
                xien,
                danDe,
                actualGdb,
                actualLoto
            };

            const evaluated = autoEvaluateRecord(baseRec, actualGdb, actualLoto);

            if (editId) {
                const idx = trackerRecords.findIndex(r => r.id === editId);
                if (idx !== -1) trackerRecords[idx] = evaluated;
            } else {
                trackerRecords.unshift(evaluated);
            }

            saveTrackerToStorage();
            renderWinLossTracker();
            toggleNewPredictionModal(false);

            const statusMsg = evaluated.status === 'WIN'
                ? `🎉 Kết quả đối soát: ĐÃ TRÚNG (${evaluated.hitDetails})!`
                : evaluated.status === 'LOSS'
                ? `Kết quả đối soát: KHÔNG TRÚNG kỳ ${date}.`
                : `Đã lưu kỳ ${date} vào danh sách Chờ Kết Quả 18:15!`;
            showToast(statusMsg, evaluated.status === 'LOSS');
        }

        function exportWinLossReport() {
            const settled = trackerRecords.filter(r => r.status === 'WIN' || r.status === 'LOSS');
            const wins = trackerRecords.filter(r => r.status === 'WIN');
            const losses = trackerRecords.filter(r => r.status === 'LOSS');
            const winRate = settled.length > 0 ? ((wins.length / settled.length) * 100).toFixed(1) : '0.0';

            const rowsText = trackerRecords.slice(0, 15).map(r => {
                const icon = r.status === 'WIN' ? '✅ ĐÃ TRÚNG' : r.status === 'LOSS' ? '❌ KHÔNG TRÚNG' : '⏳ CHỜ KQ';
                return `• ${r.date} | BTL: ${r.btl} | STL: ${(r.stl || []).join('-')} | ${icon} (${r.hitDetails})`;
            }).join('\n');

            const report =
`📊 BÁO CÁO THỐNG KÊ HIỆU SUẤT DỰ ĐOÁN XSMB AI 📊
- Tỷ lệ thắng tổng (Win Rate): ${winRate}% (${wins.length} Trúng / ${losses.length} Không Trúng)
- Chi tiết gần nhất:
${rowsText}`;

            copyTextToClipboard(report, 'Đã copy báo cáo thống kê Trúng / Không Trúng!');
        }
